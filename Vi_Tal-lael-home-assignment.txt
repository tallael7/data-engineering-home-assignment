###------------------actions for main Objectives
create S3 bucket (currently I created 1 Bucket, in the real world I will create bucket for each folder inside the current bucket (DB's, Athena_code, logs, Glue_code and more) for more comfortable versioning and automatic deletion actions)
Bucket name is ending with -dev for Managing test and production environments, in other environments\account we will have -tst and -prod
create folders inside bucket
open Athena and set query result to s3://data-engineer-assignment-tal-lael-dev/Athena_code/
run Athena code:
create database tal_lael_dw_staging LOCATION 's3://data-engineer-assignment-tal-lael-dev/dw_staging/';

create database tal_lael_dw LOCATION 's3://data-engineer-assignment-tal-lael-dev/dw/';

CREATE EXTERNAL TABLE `stocks_data`(
  `date` string COMMENT 'from deserializer', 
  `open` string COMMENT 'from deserializer', 
  `high` string COMMENT 'from deserializer', 
  `low` string COMMENT 'from deserializer', 
  `close` string COMMENT 'from deserializer', 
  `volume` string COMMENT 'from deserializer', 
  `ticker` string COMMENT 'from deserializer')
ROW FORMAT SERDE 
  'org.apache.hadoop.hive.serde2.OpenCSVSerde' 
WITH SERDEPROPERTIES ( 
  'separatorChar'=',') 
STORED AS INPUTFORMAT 
  'org.apache.hadoop.mapred.TextInputFormat' 
OUTPUTFORMAT 
  'org.apache.hadoop.hive.ql.io.HiveIgnoreKeyTextOutputFormat'
LOCATION
  's3://data-engineer-assignment-tal-lael-dev/dw_staging/'
TBLPROPERTIES (
  'classification'='csv', 
  'skip.header.line.count'='1')

CREATE EXTERNAL TABLE `avg_closing_rate`(
  `date` date, 
  `avg_closing_rate` decimal(10,5))
ROW FORMAT SERDE 
  'org.apache.hadoop.hive.ql.io.parquet.serde.ParquetHiveSerDe' 
WITH SERDEPROPERTIES ( 
  'paths'='date') 
STORED AS INPUTFORMAT 
  'org.apache.hadoop.mapred.TextInputFormat' 
OUTPUTFORMAT 
  'org.apache.hadoop.hive.ql.io.HiveIgnoreKeyTextOutputFormat'
LOCATION
  's3://data-engineer-assignment-tal-lael-dev/dw/avg_closing_rate'
TBLPROPERTIES (
  'classification'='Parquet', 
  'has_encrypted_data'='false')
